; =====================================================================
; Brymen BM78xBT via brymen-tc-bridge - SingleValue driver.
;
; Run the bridge with the default template:
;
;     python -m bridge --format "{mode} {si_value}"
;
;   e.g.   "DCV 607.8"    "DCmV 0.04530"    "RES 10250"    "DCmA 0.003456"
;
; The mode token is the LEADING letters of each line - TestController's
; SingleValue driver builds the mode exactly from those letters (unit letters
; after the number would be appended to it, so we don't emit any). The
; selector (last field of each #value row) must equal that token, so the
; correct column/unit follows the meter's rotary switch automatically.
;
; Values are sent in BASE units (si_value) and each row uses the "si" format,
; so TestController re-applies the SI prefix for display (45.30 mV -> "45.3m").
; =====================================================================

#idString Brymen,BM78xBT,
#name Brymen BM78xBT
#handle BM78x
#port 6000
#driver SingleValue

#interfaceType DMM
#interface readValue 0

; --- Values (selector = mode token emitted by the bridge) -------------
#value DCV V si DCV
#value ACV V si ACV
#value DCACV V si DCACV
#value DCmV V si DCmV
#value ACmV V si ACmV
#value DCACmV V si DCACmV
#value DCA A si DCA
#value ACA A si ACA
#value DCACA A si DCACA
#value DCmA A si DCmA
#value ACmA A si ACmA
#value DCACmA A si DCACmA
#value DCuA A si DCuA
#value ACuA A si ACuA
#value DCACuA A si DCACuA
#value RES Ohm si RES
#value CAP F si CAP
#value CONT Ohm si CONT
#value DIODE V si DIODE
#value COND S si COND
#value DUTY % si DUTY
#value LOGIC Hz si LOGIC
#value TC C si TC
#value TD C si TD
#value TCD C si TCD
#value AUTO - si AUTO
#value LoZACV V si LoZACV
#value LoZDCV V si LoZDCV
#value VFDAC V si VFDAC
#value VFDHz Hz si VFDHz
#value LINEHz Hz si LINEHz
#value PCT % si PCT
#value EFLO - si EFLO
#value EFHI - si EFHI

; --- Overload / error states: the bridge emits "<mode> OL " (trailing space)
; TC's valueText handler strips the token, rebuilds the mode from the
; remaining letters and reads one char past the token; a bare / last-token
; "OL" drops the socket, a trailing unit pollutes the mode. The trailing
; space keeps the mode clean ("DCV OL " -> mode "DCV"). The "-OL" row is
; retained for completeness but TC renders negative overload as +inf.
#valueText OL OL
#valueText -OL "- OL"
#valueText Auto Auto
#valueText InEr InEr
#valueText EF-H EF-H
#valueText EF-L EF-L
#valueText "-----" "-----"
#valueText "----" "----" ; temperature overload (meter shows 4 dashes)
; Dedicated gap token for the bridge's data-gap keep-alive (NOT a meter state).
#valueText ? ?
